cmake_minimum_required(VERSION 3.22)
project(zara_offline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  foreach(flag LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL GGML_NATIVE GGML_OPENMP GGML_LLAMAFILE GGML_CPU_KLEIDIAI)
    set(${flag} OFF CACHE BOOL "" FORCE)
  endforeach()
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../llama.cpp" "${CMAKE_CURRENT_BINARY_DIR}/llama")
  if(ANDROID)
    add_library(zara_offline SHARED offline_jni.cpp inference.cpp)
    target_link_libraries(zara_offline PRIVATE llama)
    target_link_options(zara_offline PRIVATE "-Wl,-z,max-page-size=16384")
  else()
    add_executable(zara_offline_smoke smoke.cpp inference.cpp)
    target_link_libraries(zara_offline_smoke PRIVATE llama)
  endif()
